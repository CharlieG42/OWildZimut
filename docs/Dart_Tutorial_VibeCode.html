1.1 Variables et Types

📌 Syntaxe de Base

// Déclaration de variables
var name = 'Dart';          // Type inféré

num value = 42;            // int ou double

✅ Bonnes Pratiques

  • Préférer le typage explicite pour la lisibilité
  • Utiliser final pour les variables immuables
  • Utiliser const pour les constantes connues à la compilation

❌ Erreurs Fréquentes

// ❌ MAUVAIS
var isEnabled = True;   // True/False = syntaxe Python !
var count = null;      // null sans typage nullable

// ✅ BON
bool isEnabled = true; // true/false en minuscules
int? count = null;     // Typage nullable

1.2 Opérateurs

📌 Opérateurs Arithmétiques

int a = 10, b = 3;
print(a + b);   // 13
print(a - b);   // 7
print(a * b);   // 30
print(a / b);   // 3.333...
print(a ~/ b);  // 3 (division entière)
print(a % b);   // 1 (modulo)

📌 Opérateurs de Comparaison

print(a == b);   // false
print(a != b);   // true
print(a > b);    // true
print(a >= b);   // true
print(a is int); // true (test de type)
print(a is! int); // false (négation)

📌 Opérateurs Logiques

bool x = true, y = false;
print(x && y); // false (ET)
print(x || y); // true  (OU)
print(!x);     // false (NON)

📌 Opérateurs Null-Aware (Très Important !)

String? name; // Variable nullable

// Accès conditionnel
print(name?.length); // null si name est null (au lieu de lancer une exception)

// Opérateur de coalescence null
print(name ?? 'Unknown'); // 'Unknown' si name est null

// Assertion non-null
print(name!.length); // Lance une exception si name est null

📌 Paramètres

// Paramètres nommés

  print('Hello $name, age $age');
}
greet(name: 'Alice', age: 25); // Appel avec paramètres nommés

// Paramètres optionnels positionnels
void greet([String name = 'User', int age = 0]) {}
greet('Bob'); // age prendra la valeur par défaut

// Paramètres nommés obligatoires
void greet({required String name, int age = 0}) {}
greet(name: 'Charlie'); // age est optionnel, name est obligatoire

2.2 Constructeurs

📌 Constructeurs Primaires

// Syntaxe courte (recommandée)
class Person {
  final String name;
  final int age;

  Person(this.name, this.age);
}

// Syntaxe longue
class Person {
  final String name;
  final int age;

  Person(String name, int age) {
    this.name = name;
    this.age = age;
  }
}

📌 Constructeurs Nommés

class Person {
  final String name;
  final int age;

  Person(this.name, this.age);

  // Constructeur nommé
  Person.fromJson(Map<String, dynamic> json)
    : name = json['name'],
      age = json['age'];

  // Constructeur nommé avec logique
  Person.baby(String name) : this(name, 0);
}

// Utilisation
var person1 = Person('Alice', 25);
var person2 = Person.fromJson({'name': 'Bob', 'age': 30});
var person3 = Person.baby('Charlie');

❌ Erreur Fréquente : Syntaxe Invalide

// ❌ MAUVAIS - Cette syntaxe n'existe pas en Dart !
static const Person.fromJson(Map<String, dynamic> json)
  : name = json['name'],
    age = json['age'];

// ✅ BON - Utiliser factory si nécessaire, ou un constructeur classique
Person.fromJson(Map<String, dynamic> json)
  : name = json['name'],
    age = json['age'];

// Ou avec factory (si création conditionnelle)
factory Person.fromJson(Map<String, dynamic> json) {
  return Person(json['name'], json['age']);
}

2.3 Factory Constructors

📌 Quand utiliser factory ?

  • Retourner une instance existante (cache)
  • Retourner un sous-type
  • Logique complexe de création

📌 Exemples

class Logger {
  static final _instance = Logger._internal();

  // Factory pour singleton
  factory Logger() => _instance;

  // Constructeur privé
  Logger._internal();
}

class Animal {
  final String type;

  Animal(this.type);

  // Factory pour retourner des sous-types
  factory Animal.fromType(String type) {
    switch (type) {
      case 'dog': return Dog();
      case 'cat': return Cat();
      default: return Animal(type);
    }
  }
}

class Dog extends Animal {
  Dog() : super('dog');
}

class Cat extends Animal {
  Cat() : super('cat');
}

2.4 Null Safety (TRÈS IMPORTANT !)

📌 Concepts Clés

Dart a un système de null safety strict : - Type : Non-nullable (ne peut pas être null) - Type? : Nullable (peut être null)

📌 Exemples

// ❌ MAUVAIS - Erreur de compilation
String name = null;

// ✅ BON - Nullable
String? name = null;

// Accès conditionnel
name?.length; // Retourne null si name est null

// Opérateur de coalescence
String displayName = name ?? 'Unknown';

// Assertion non-null (à utiliser avec prudence !)
print(name!.length); // Lance une exception si name est null

📌 Constructeurs et Null Safety

class Person {
  final String name;
  final int? age; // Nullable

  // ❌ MAUVAIS - age est obligatoire mais peut être null
  Person(this.name, this.age);

  // ✅ BON - age est optionnel
  Person(this.name, [this.age]);

  // ✅ BON - age est nullable et optionnel
  Person(this.name, {this.age});
}

class Temperature {
  double _celsius = 0;

  // Getter
  double get celsius => _celsius;

  // Setter avec validation
  set celsius(double value) {
    if (value < -273.15) {
      throw ArgumentError('Température trop basse');
    }
    _celsius = value;
  }

  // Getter calculé
  double get fahrenheit => _celsius * 9/5 + 32;

  // Setter calculé
  set fahrenheit(double value) {
    _celsius = (value - 32) * 5/9;
  }
}

2.6 toString(), ==, et hashCode

📌 Méthodes Spéciales

class Person {
  final String name;
  final int age;

  Person(this.name, this.age);

  // toString() pour l'affichage
  @override
  String toString() => 'Person(name: $name, age: $age)';

  // == pour la comparaison
  @override
  bool operator ==(Object other) =>
    identical(this, other) ||
    other is Person &&
    other.name == name &&
    other.age == age;

  // hashCode pour les collections (Set, Map)
  @override
  int get hashCode => Object.hash(name, age);
}

// Utilisation
var p1 = Person('Alice', 25);
var p2 = Person('Alice', 25);
print(p1 == p2); // true (grâce à operator ==)
print(p1); // Person(name: Alice, age: 25) (grâce à toString)

2.7 Exercices Pratiques

📌 Exercice 1 : Classe BankAccount

class BankAccount {
  final String owner;
  double _balance;

  BankAccount(this.owner, [double initialBalance = 0]) : _balance = initialBalance;

  // Getter
  double get balance => _balance;

  // Méthodes
  void deposit(double amount) {
    if (amount <= 0) throw ArgumentError('Montant invalide');
    _balance += amount;
  }

  void withdraw(double amount) {
    if (amount <= 0) throw ArgumentError('Montant invalide');
    if (amount > _balance) throw Exception('Solde insuffisant');
    _balance -= amount;
  }

  @override
  String toString() => 'BankAccount(owner: $owner, balance: $_balance)';
}

📌 Exercice 2 : Classe Point avec Null Safety

class Point {
  final double x;
  final double y;

  Point(this.x, this.y);

  // Factory pour créer depuis un Map
  factory Point.fromJson(Map<String, dynamic> json) {
    return Point(
      json['x']?.toDouble() ?? 0.0,
      json['y']?.toDouble() ?? 0.0,
    );
  }

  // Distance entre deux points
  double distanceTo(Point other) {
    final dx = x - other.x;
    final dy = y - other.y;
    return sqrt(dx * dx + dy * dy);
  }

  @override
  bool operator ==(Object other) =>
    identical(this, other) ||
    other is Point &&
    other.x == x &&
    other.y == y;

  @override
  int get hashCode => Object.hash(x, y);

  @override
  String toString() => 'Point($x, $y)';
}

List<int> evens = numbers.where((n) => n % 2 == 0).toList(); // [2, 4]

int? firstEven = numbers.firstWhere((n) => n % 2 == 0); // 2

bool hasEven = numbers.any((n) => n % 2 == 0); // true
bool allEven = numbers.every((n) => n % 2 == 0); // false

List<double> calculateStats(List<double> numbers) {
  if (numbers.isEmpty) return [0, 0, 0];

  double sum = numbers.reduce((a, b) => a + b);
  double average = sum / numbers.length;
  double max = numbers.reduce((a, b) => a > b ? a : b);
  double min = numbers.reduce((a, b) => a < b ? a : b);

  return [average, max, min];
}

Future<String> fetchData() async {
  try {
    // Simulation d'une erreur
    if (DateTime.now().second % 2 == 0) {
      return 'Données reçues';
    } else {
      throw Exception('Erreur de réseau');
    }
  } catch (e) {
    print('Erreur: $e');
    rethrow; // Relancer l'exception
  }
}

// Utilisation avec catch
Future<void> useData() async {
  try {
    String data = await fetchData();
    print(data);
  } catch (e) {
    print('Erreur capturée: $e');
  }
}

4.2 Streams

📌 Concepts de Base

// Créer un Stream
Stream<int> countStream(int max) async* {
  for (int i = 1; i <= max; i++) {
    await Future.delayed(Duration(seconds: 1));
    yield i;
  }
}

// Utiliser un Stream
Future<void> listenToStream() async {
  await for (int value in countStream(5)) {
    print(value); // 1, 2, 3, 4, 5 (une valeur par seconde)
  }
}

📌 Méthodes des Streams

Stream<int> numbers = Stream.fromIterable([1, 2, 3, 4, 5]);

4.5 Exercices Pratiques

📌 Exercice 1 : Fetch API avec Retry

Future<String> fetchWithRetry(String url, {int maxRetries = 3}) async {
  for (int i = 0; i < maxRetries; i++) {
    try {
      // Simulation de fetch
      await Future.delayed(Duration(seconds: 1));
      if (i < maxRetries - 1) {
        throw Exception('Échec temporaire');
      }
      return 'Données de $url';
    } catch (e) {
      print('Tentative $i échouée: $e');
      if (i == maxRetries - 1) rethrow;
    }
  }
  throw Exception('Toutes les tentatives ont échoué');
}

// Utilisation
Future<void> main() async {
  try {
    String data = await fetchWithRetry('https://api.example.com');
    print(data);
  } catch (e) {
    print('Erreur finale: $e');
  }
}

❌ Erreurs Fréquentes et Corrections

📌 Tableau Récapitulatif

Erreur Mauvais Bon Explication
Booléens Python True, False true, false Dart utilise des minuscules
Constructeur factory static const X.nom() : ... factory X.nom() { return ...; } Syntaxe Dart pour les constructeurs factory
Champ non-nullable required this.x (si x peut être null) this.x (nullable) Vérifier si le champ est toujours présent
Formatage numérique ${x:.2f} ${x.toStringAsFixed(2)} Méthode Dart pour le formatage
Null safety String x = null; String? x = null; Dart exige le typage nullable
Accès conditionnel x.length (si x peut être null) x?.length Évite les exceptions
Opérateur de coalescence x != null ? x : 'default' x ?? 'default' Plus concis
Constructeur constant const X(this.x) avec x non-final const X(this.x) avec x final Tous les champs doivent être final
